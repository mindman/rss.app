'' : s) .replace(/&/g, '&').replace(//g, '>') .replace(/"/g, '"').replace(/'/g, '''); } function safeUrl(u) { u = String(u || '').trim(); return /^https?:\/\//i.test(u) ? u : ''; } function truncate(s, n) { s = String(s || ''); if (n <= 0) return ''; var chars = Array.from(s); return chars.length > n ? chars.slice(0, n).join('') + '…' : s; } function pad(n) { return (n < 10 ? '0' : '') + n; } function formatDate(iso, fmt) { if (!iso) return ''; var d = new Date(iso); if (isNaN(d.getTime())) return ''; var ymd = d.getFullYear() + '/' + pad(d.getMonth() + 1) + '/' + pad(d.getDate()); if (fmt === 'date') return ymd; if (fmt === 'datetime') return ymd + ' ' + pad(d.getHours()) + ':' + pad(d.getMinutes()); var sec = (Date.now() - d.getTime()) / 1000; if (sec < 60) return 'たった今'; if (sec < 3600) return Math.floor(sec / 60) + '分前'; if (sec < 86400) return Math.floor(sec / 3600) + '時間前'; if (sec < 86400 * 7) return Math.floor(sec / 86400) + '日前'; return ymd; } function hostOf(u) { var m = String(u || '').match(/^https?:\/\/([^\/?#]+)/i); return m ? m[1].replace(/^www\./, '') : ''; } /** 設定値をもう一度クライアント側でも検証する（CSS に入るため） */ function normalize(c) { c = c || {}; var n = function (v, lo, hi, d) { v = parseInt(v, 10); return isNaN(v) ? d : Math.max(lo, Math.min(hi, v)); }; return { theme: THEMES.indexOf(c.theme) >= 0 ? c.theme : 'list', title: String(c.title || ''), count: n(c.count, 1, 50, 6), columns: n(c.columns, 1, 4, 3), showHeader: c.showHeader !== false, showImage: c.showImage !== false, showDate: c.showDate !== false, showDesc: c.showDesc !== false, descLength: n(c.descLength, 0, 300, 80), dateFormat: ['relative', 'date', 'datetime'].indexOf(c.dateFormat) >= 0 ? c.dateFormat : 'relative', newTab: c.newTab !== false, accent: /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.test(c.accent || '') ? c.accent : '#2563eb', radius: n(c.radius, 0, 32, 10), font: FONTS[c.font] ? c.font : 'system', scheme: ['light', 'dark', 'auto'].indexOf(c.scheme) >= 0 ? c.scheme : 'light', transparent: !!c.transparent, speed: n(c.speed, 10, 200, 50) }; } // ---------- パーツ ---------- function linkAttrs(item, c) { var href = safeUrl(item.link); return (href ? 'href="' + esc(href) + '"' : '') + (c.newTab ? ' target="_blank" rel="noopener noreferrer"' : ' target="_top"'); } function img(item, cls, c) { var src = safeUrl(item.image); if (!c.showImage) return ''; if (!src) { var label = hostOf(item.link) || 'RSS'; return '

'; } function meta(item, c) { var parts = []; if (c.showDate && item.date) { parts.push('' + esc(formatDate(item.date, c.dateFormat)) + ''); } if (item.author) parts.push('' + esc(item.author) + ''); return parts.length ? '
' + parts.join('·') + '
' : ''; } function desc(item, c, len) { if (!c.showDesc) return ''; var t = truncate(item.desc, len == null ? c.descLength : len); return t ? '

' + esc(t) + '

' : ''; } function title(item, tag) { return '<' + tag + ' class="rw-title">' + esc(item.title) + ''; } function header(feed, c) { if (!c.showHeader) return ''; var name = c.title || feed.title || hostOf(feed.link) || 'RSS'; var href = safeUrl(feed.link); var inner = '' + esc(name) + ''; return '

'; } // ---------- テーマ別テンプレート ---------- var T = {}; T.list = function (items, c) { return '

'; }).join('') + '

'; var showHeader = c.theme !== 'ticker'; return '
' + (showHeader ? header(feed, c) : '') + body + '
'; } function wire(root, c) { // 読み込めなかった画像はプレースホルダーに置き換える Array.prototype.forEach.call(root.querySelectorAll('img'), function (im) { im.addEventListener('error', function () { var box = im.parentNode; if (!box) return; box.classList.add('rw-ph'); box.innerHTML = '•'; }, { once: true }); }); var car = root.querySelector('.rw-carousel'); if (car) { var track = car.querySelector('.rw-track'); var step = function (dir) { var slide = track.querySelector('.rw-slide'); var w = slide ? slide.getBoundingClientRect().width + 14 : track.clientWidth * 0.8; track.scrollBy({ left: dir * w, behavior: 'smooth' }); }; var sync = function () { car.querySelector('.rw-prev').disabled = track.scrollLeft <= 2; car.querySelector('.rw-next').disabled = track.scrollLeft + track.clientWidth >= track.scrollWidth - 2; }; car.querySelector('.rw-prev').addEventListener('click', function () { step(-1); }); car.querySelector('.rw-next').addEventListener('click', function () { step(1); }); track.addEventListener('scroll', sync, { passive: true }); setTimeout(sync, 0); } // ティッカーは 1 列の幅から「speed px/秒」になるようにアニメーション時間を決める var row = root.querySelector('.rw-trow'); if (row) { var fit = function () { var w = row.getBoundingClientRect().width; if (w) root.querySelector('.rw-tk').style.setProperty('--rw-dur', Math.max(5, w / c.speed).toFixed(1) + 's'); }; fit(); setTimeout(fit, 500); } } /** * 要素にウィジェットを描画する。Shadow DOM を使うので埋め込み先の CSS と干渉しない。 * 同じ要素に再度呼ぶと描き直す（エディタのプレビューで使用）。 */ function mount(el, feed, config, error) { var root = el.attachShadow ? (el.shadowRoot || el.attachShadow({ mode: 'open' })) : el; root.innerHTML = '' + renderHtml(feed, config, error); wire(root, normalize(config)); el.setAttribute('data-rw-mounted', '1'); return root; } return { mount: mount, renderHtml: renderHtml, normalize: normalize, themes: THEMES, version: '1.0.0' };
